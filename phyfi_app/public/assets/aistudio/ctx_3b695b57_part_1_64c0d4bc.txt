package com.ug.phyfi_router.audio

import android.annotation.SuppressLint
import android.media.AudioFormat
import android.media.AudioRecord
import android.media.MediaRecorder
import android.os.Process
import android.util.Log

/**
 * Gerencia o hardware de microfone, operando em taxa de amostragem padrão de áudio digital (44.1kHz).
 * Garante que a thread de captura rode com prioridade de áudio (URGENT_AUDIO) do Linux Kernel.
 */
class AudioCaptureManager(
    private val onAudioDataReceived: (ShortArray) -> Unit
) {
    private var audioRecord: AudioRecord? = null
    private var isRecording = false
    private var captureThread: Thread? = null

    // 44100Hz é o padrão universal que engloba a faixa que precisamos (1kHz - 4kHz)
    private val sampleRate = 44100
    private val channelConfig = AudioFormat.CHANNEL_IN_MONO
    private val audioFormat = AudioFormat.ENCODING_PCM_16BIT
    
    private val minBufferSize = AudioRecord.getMinBufferSize(sampleRate, channelConfig, audioFormat)

    @SuppressLint("MissingPermission") // Permissão já garantida pelo PermissionManager
    fun startCapture() {
        if (isRecording) return

        if (minBufferSize == AudioRecord.ERROR_BAD_VALUE || minBufferSize == AudioRecord.ERROR) {
            Log.e("AudioCapture", "Configuração de áudio não suportada pelo hardware.")
            return
        }

        audioRecord = AudioRecord(
            MediaRecorder.AudioSource.MIC,
            sampleRate,
            channelConfig,
            audioFormat,
            minBufferSize * 2 // Buffer duplo para evitar underflow
        )

        if (audioRecord?.state != AudioRecord.STATE_INITIALIZED) {
            Log.e("AudioCapture", "Falha ao inicializar o AudioRecord.")
            return
        }

        isRecording = true
        audioRecord?.startRecording()

        captureThread = Thread {
            // Eleva a prioridade da thread a nível de SO para evitar interrupções do Scheduler
            Process.setThreadPriority(Process.THREAD_PRIORITY_URGENT_AUDIO)
            readAudioLoop()
        }.apply {
            name = "PhyFi-AudioCaptureThread"
            start()
        }
    }

    private fun readAudioLoop() {
        // Buffer dimensionado para blocos rápidos (reduz latência física de captura)
        val buffer = ShortArray(minBufferSize)

        while (isRecording && !Thread.currentThread().isInterrupted) {
            val readResult = audioRecord?.read(buffer, 0, buffer.size) ?: 0
            if (readResult > 0) {
                // Passa a cópia do array preenchido para não sofrer mutação durante a FFT
                onAudioDataReceived(buffer.copyOf(readResult))
            }
        }
    }

    fun stopCapture() {
        isRecording = false
        captureThread?.interrupt()
        captureThread = null
        
        audioRecord?.apply {
            if (recordingState == AudioRecord.RECORDSTATE_RECORDING) {
                stop()
            }
            release()
        }
        audioRecord = null
    }
}

package com.ug.phyfi_router.audio

import be.tarsos.dsp.util.fft.FFT
import be.tarsos.dsp.util.fft.HannWindow
import kotlin.math.abs

/**
 * Responsável pela extração esteganográfica baseada em Dicionário Hexadecimal.
 * Filtra frequências vocais (1kHz-4kHz), rejeita ruído e reconstrói pacotes de 16-bits.
 */
class AudioDecoder(
    private val sampleRate: Float = 44100f,
    private val onPayloadDecoded: (Short) -> Unit
) {
    // Janela de FFT (Potência de 2 estrita). 2048 samples a 44.1kHz gera latência física de ~46ms.
    private val fftSize = 2048
    private val fft = FFT(fftSize, HannWindow())
    private val floatBuffer = FloatArray(fftSize)
    private val amplitudes = FloatArray(fftSize / 2)

    private val bufferQueue = mutableListOf<Short>()

    // Dicionário de Frequências (0x0 a 0xF) espalhadas na zona de ouro psicoacústica (1kHz a ~3.4kHz).
    // Estes valores devem casar perfeitamente com os "Soundbanks" do script Haxe na engine.
    private val frequencyBins = floatArrayOf(
        1100f, 1250f, 1400f, 1550f, 1700f, 1850f, 2000f, 2150f,
        2300f, 2450f, 2600f, 2750f, 2900f, 3050f, 3200f, 3350f
    )

    // Acumulador: são necessários 4 "bips" hexadecimais (4 bits cada) para formar o pacote Kine-Fi de 16-bits
    private val hexBuffer = ByteArray(4)
    private var hexIndex = 0

    fun processAudio(shortBuffer: ShortArray) {
        bufferQueue.addAll(shortBuffer.toList())

        // Processa todos os chunks disponíveis que atingiram a potência de 2
        while (bufferQueue.size >= fftSize) {
            val chunk = bufferQueue.take(fftSize)
            bufferQueue.subList(0, fftSize).clear()
            executeFFT(chunk)
        }
    }

    private fun executeFFT(chunk: List<Short>) {
        // O algoritmo TarsosDSP exige float normalizado entre -1.0 e 1.0
        for (i in chunk.indices) {
            floatBuffer[i] = chunk[i] / 32768f
        }

        fft.forwardTransform(floatBuffer)
        fft.modulus(floatBuffer, amplitudes)

        var maxAmp = 0f
        var targetFreq = 0f

        // Varredura para encontrar a frequência dominante
        for (i in amplitudes.indices) {
            val currentFreq = (i * sampleRate) / fftSize
            
            // Bypass físico: Analisa apenas a faixa alvo. Ignora graves (passos, motor) e agudos descartados.
            if (currentFreq in 1000f..4000f) {
                if (amplitudes[i] > maxAmp) {
                    maxAmp = amplitudes[i]
                    targetFreq = currentFreq
                }
            }
        }

        // Noise Gate matemático: Ignora ruídos ambientes normais. (Limiar exigirá fine-tuning em testes físicos)
        if (maxAmp > 0.05f) {
            mapFrequencyToHex(targetFreq)?.let { hexValue ->
                accumulateHex(hexValue)
            }
        }
    }

    private fun mapFrequencyToHex(freq: Float): Byte? {
        val tolerance = 60f // Margem de erro em Hz devido à compressão de áudio
        for (i in frequencyBins.indices) {
            if (abs(freq - frequencyBins[i]) <= tolerance) {
                return i.toByte()
            }
        }
        return null
    }

    private fun accumulateHex(hexValue: Byte) {
        hexBuffer[hexIndex] = hexValue
        hexIndex++

        if (hexIndex == 4) {
            val payload = buildShortFromHex()
            onPayloadDecoded(payload)
            hexIndex = 0
        }
    }

    private fun buildShortFromHex(): Short {
        val h1 = (hexBuffer[0].toInt() and 0x0F) shl 12
        val h2 = (hexBuffer[1].toInt() and 0x0F) shl 8
        val h3 = (hexBuffer[2].toInt() and 0x0F) shl 4
        val h4 = hexBuffer[3].toInt() and 0x0F
        return (h1 or h2 or h3 or h4).toShort()
    }
}

@SuppressLint("MissingPermission")
class BluetoothHidManager(private val context: Context) : BluetoothProfile.ServiceListener {

    private val bluetoothManager =
            context.getSystemService(Context.BLUETOOTH_SERVICE) as BluetoothManager
    private val bluetoothAdapter = bluetoothManager.adapter

    private var hidDevice: BluetoothHidDevice? = null
    private var connectedHost: BluetoothDevice? = null
    private var isRegistered = false

    private val scope = CoroutineScope(Dispatchers.IO)

    private val callback =
            object : BluetoothHidDevice.Callback() {
                override fun onAppStatusChanged(
                        pluggedDevice: BluetoothDevice?,
                        registered: Boolean
                ) {
                    isRegistered = registered
                    if (registered) {
                        logAsync("SDP HID Registrado. Tentando forçar conexão com o PC...")
                        connectToHost()
                    }
                }

                override fun onConnectionStateChanged(device: BluetoothDevice, state: Int) {
                    when (state) {
                        BluetoothProfile.STATE_CONNECTED -> {
                            connectedHost = device
                            RouterState.setBluetoothStatus(true)
                            logAsync("Hospedeiro conectado e aceitando inputs: ${device.address}")
                        }
                        BluetoothProfile.STATE_DISCONNECTED -> {
                            connectedHost = null
                            RouterState.setBluetoothStatus(false)
                            logAsync("Hospedeiro desconectado.")
                        }
                    }
                }
            }

    fun initialize() {
        if (bluetoothAdapter?.isEnabled != true) {
            logAsync("ERRO: Bluetooth desligado no SO.")
            return
        }
        bluetoothAdapter.getProfileProxy(context, this, BluetoothProfile.HID_DEVICE)
    }

    override fun onServiceConnected(profile: Int, proxy: BluetoothProfile?) {
        if (profile == BluetoothProfile.HID_DEVICE) {
            hidDevice = proxy as BluetoothHidDevice
            registerApp()
        }
    }

    override fun onServiceDisconnected(profile: Int) {
        if (profile == BluetoothProfile.HID_DEVICE) {
            hidDevice = null
        }
    }

    private fun registerApp() {
        hidDevice?.registerApp(
                HidConstants.SDP_SETTINGS,
                null,
                null,
                context.mainExecutor,
                callback
        )
    }

    private fun connectToHost() {
        if (hidDevice == null) return
        val bondedDevices = bluetoothAdapter?.bondedDevices ?: return

        // Tenta iniciar a conexão de dados com os dispositivos pareados.
        // O OS ignorará os que não estiverem visíveis.
        for (device in bondedDevices) {
            hidDevice?.connect(device)
        }
    }

    fun sendKeyboardReport(report: ByteArray) {
        if (connectedHost == null) return
        hidDevice?.sendReport(connectedHost, HidConstants.REPORT_ID_KEYBOARD, report)
    }

    fun shutdown() {
        if (isRegistered) {
            hidDevice?.unregisterApp()
        }
        hidDevice?.let { bluetoothAdapter?.closeProfileProxy(BluetoothProfile.HID_DEVICE, it) }
    }

    private fun logAsync(msg: String) {
        scope.launch { RouterState.log(msg) }
    }
}

import com.ug.phyfi_router.protocol.PhyFiPayload
import kotlin.random.Random
import kotlinx.coroutines.CoroutineScope

/**
 * Traduz a intenção lógica (16-bits) em comandos mecânicos (Bluetooth HID), injetando ruído
 * anatômico (Jitter) para evasão de heurísticas anti-trapaça.
 */
class KineFiTranslator(private val hidManager: BluetoothHidManager) {
    private val scope = CoroutineScope(Dispatchers.IO)

    fun processIncomingPayload(payload: PhyFiPayload) {
        val report = buildHidReport(payload.intent)

        scope.launch {
            // Jitter KeyDown: flutuação de 5ms a 22ms exigida na documentação
            val downJitter = Random.nextLong(5, 23)
            delay(downJitter)
            hidManager.sendKeyboardReport(report)

            // Duração do contato mecânico simulado
            val pressDuration = Random.nextLong(30, 80)
            delay(pressDuration)

            // Jitter KeyUp: imperfeição na liberação da tecla
            val upJitter = Random.nextLong(5, 23)
            delay(upJitter)
            hidManager.sendKeyboardReport(emptyReport())
        }
    }

    private fun buildHidReport(intent: Byte): ByteArray {
        val report = ByteArray(8)
        // Padrão de relatório HID Keyboard:
        // [0: Modificador, 1: Reservado, 2: Tecla1, 3: Tecla2, 4: Tecla3, 5: Tecla4, 6: Tecla5, 7:
        // Tecla6]

        // Estruturação do Chord Obscuro: Left Ctrl (0x01) + F15 (0x6A) + Tecla Dinâmica
        report[0] = 0x01
        report[2] = 0x6A
        report[3] = intent

        return report
    }

    private fun emptyReport(): ByteArray {
        return ByteArray(8)
    }
}

class PermissionManager(private val activity: AppCompatActivity) {
    private val permissionsToRequest: Array<String> = getPermissionsForCurrentSdk()
    private var onResultCallback: ((Boolean) -> Unit)? = null

    // O registro OBRIGATORIAMENTE deve acontecer durante o onCreate da Activity
    private val requestLauncher =
            activity.registerForActivityResult(
                    ActivityResultContracts.RequestMultiplePermissions()
            ) { permissionsMap ->
                val allGranted =
                        permissionsToRequest.all { perm ->
                            permissionsMap[perm] == true ||
                                    ContextCompat.checkSelfPermission(activity, perm) ==
                                            PackageManager.PERMISSION_GRANTED
                        }
                onResultCallback?.invoke(allGranted)
            }

    fun checkAndRequest(onResult: (Boolean) -> Unit) {
        this.onResultCallback = onResult
        val missingPermissions =
                permissionsToRequest.filter {
                    ContextCompat.checkSelfPermission(activity, it) !=
                            PackageManager.PERMISSION_GRANTED
                }

        if (missingPermissions.isEmpty()) {
            onResult(true)
        } else {
            requestLauncher.launch(missingPermissions.toTypedArray())
        }
    }

    private fun getPermissionsForCurrentSdk(): Array<String> {
        val list = mutableListOf(Manifest.permission.RECORD_AUDIO)

        if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.S) {
            list.add(Manifest.permission.BLUETOOTH_CONNECT)
            list.add(Manifest.permission.BLUETOOTH_ADVERTISE)
            list.add(Manifest.permission.BLUETOOTH_SCAN)
        } else {
            list.add(Manifest.permission.BLUETOOTH)
            list.add(Manifest.permission.BLUETOOTH_ADMIN)
        }

        return list.toTypedArray()
    }
}

import kotlinx.coroutines.launch
import org.java_websocket.client.WebSocketClient
import org.java_websocket.handshake.ServerHandshake
import org.json.JSONObject

class SignalingClient(
        private val roomId: String,
        private val onStatus: (String) -> Unit,
        private val onMessageReceived: (JSONObject) -> Unit
) {
    private var webSocket: WebSocketClient? = null
    private val serverUrl = "wss://ug-signaler.onrender.com"

    private var isIntentionalDisconnect = false
    private val clientScope = CoroutineScope(Dispatchers.IO)

    fun connect() {
        isIntentionalDisconnect = false
        val uri = URI(serverUrl)

        webSocket =
                object : WebSocketClient(uri) {
                    override fun onOpen(handshakedata: ServerHandshake?) {
                        onStatus("Signaler: Conectado. Sala: $roomId")
                        joinRoom()
                    }

                    override fun onMessage(message: String?) {
                        if (message == null) return
                        try {
                            val json = JSONObject(message)
                            onMessageReceived(json)
                        } catch (e: Exception) {
                            Log.e("SignalingClient", "Falha no Parse JSON: $message")
                        }
                    }

                    override fun onClose(code: Int, reason: String?, remote: Boolean) {
                        if (!isIntentionalDisconnect) {
                            onStatus("Signaler Derrubado pelo SO ($code). Reconectando...")
                            attemptReconnect()
                        } else {
                            onStatus("Signaler: Sessão encerrada.")
                        }
                    }

                    override fun onError(ex: Exception?) {
                        Log.e("SignalingClient", "Erro WebSocket", ex)
                    }
                }

        if (uri.scheme.equals("wss", ignoreCase = true)) {
            try {
                val factory = SSLSocketFactory.getDefault() as SSLSocketFactory
                webSocket?.setSocketFactory(factory)
            } catch (e: Exception) {
                onStatus("Signaler Erro: Falha ao estabelecer SSL")
            }
        }

        webSocket?.connect()
    }

    private fun attemptReconnect() {
        clientScope.launch {
            delay(3000) // Backoff estático de 3 segundos
            if (!isIntentionalDisconnect) {
                connect()
            }
        }
    }

    private fun joinRoom() {
        val payload = JSONObject()
        payload.put("type", "join")
        send(payload)
    }

    fun send(payload: JSONObject) {
        payload.put("room", roomId)
        if (webSocket?.isOpen == true) {
            webSocket?.send(payload.toString())
        }
    }

    fun disconnect() {
        isIntentionalDisconnect = true
        webSocket?.close()
    }
}

class WebRTCManager(
    private val context: Context,
    private val signalingClient: SignalingClient,
    private val onDataReceived: (ByteArray) -> Unit
) {
    private var peerConnectionFactory: PeerConnectionFactory? = null
    private var peerConnection: PeerConnection? = null
    private var dataChannel: DataChannel? = null

    init {
        initializeFactory()
        createPeerConnection()
    }

    private fun initializeFactory() {
        val initializationOptions = PeerConnectionFactory.InitializationOptions.builder(context)
            .setEnableInternalTracer(false)
            .createInitializationOptions()
        PeerConnectionFactory.initialize(initializationOptions)

        val options = PeerConnectionFactory.Options()
        peerConnectionFactory = PeerConnectionFactory.builder()
            .setOptions(options)
            .createPeerConnectionFactory()
    }

    private fun createPeerConnection() {
        val iceServers = listOf(
            PeerConnection.IceServer.builder("stun:stun.l.google.com:19302").createIceServer()
        )
        val rtcConfig = PeerConnection.RTCConfiguration(iceServers)
        
        peerConnection = peerConnectionFactory?.createPeerConnection(rtcConfig, object : PeerConnection.Observer {
            override fun onIceCandidate(candidate: IceCandidate?) {
                if (candidate == null) return
                val payload = JSONObject()
                payload.put("type", "candidate")
                payload.put("sdpMid", candidate.sdpMid)
                payload.put("sdpMLineIndex", candidate.sdpMLineIndex)
                payload.put("candidate", candidate.sdp)
                signalingClient.send(payload)
            }

            override fun onDataChannel(dc: DataChannel?) {
                Log.i("WebRTC", "DataChannel remoto estabelecido")
                setupDataChannel(dc)
            }

            override fun onIceConnectionChange(state: PeerConnection.IceConnectionState?) {
                when (state) {
                    PeerConnection.IceConnectionState.CONNECTED, 
                    PeerConnection.IceConnectionState.COMPLETED -> RouterState.setWebRtcStatus(true)
                    PeerConnection.IceConnectionState.DISCONNECTED,
                    PeerConnection.IceConnectionState.FAILED,
                    PeerConnection.IceConnectionState.CLOSED -> RouterState.setWebRtcStatus(false)
                    else -> {}
                }
            }

            override fun onSignalingChange(p0: PeerConnection.SignalingState?) {}
            override fun onIceConnectionReceivingChange(p0: Boolean) {}
            override fun onIceGatheringChange(p0: PeerConnection.IceGatheringState?) {}
            override fun onIceCandidatesRemoved(p0: Array<out IceCandidate>?) {}
            override fun onAddStream(p0: MediaStream?) {}
            override fun onRemoveStream(p0: MediaStream?) {}
            override fun onRenegotiationNeeded() {}
            override fun onAddTrack(p0: RtpReceiver?, p1: Array<out MediaStream>?) {}
        })
    }

    fun createLocalDataChannel() {
        if (dataChannel != null) return
        val init = DataChannel.Init().apply {
            ordered = false
            maxRetransmits = 0
        }
        val dc = peerConnection?.createDataChannel("phyfi_udp_channel", init)
        setupDataChannel(dc)
    }

    private fun setupDataChannel(dc: DataChannel?) {
        dc?.registerObserver(DataChannelObserverAdapter(onDataReceived))
        dataChannel = dc
    }

    fun processSignalingMessage(json: JSONObject) {
        when (json.optString("type")) {
            "offer" -> handleOffer(json)
            "answer" -> handleAnswer(json)
            "candidate" -> handleCandidate(json)
        }
    }

    private fun handleOffer(json: JSONObject) {
        val sdp = SessionDescription(SessionDescription.Type.OFFER, json.getString("sdp"))
        peerConnection?.setRemoteDescription(SimpleSdpObserver(), sdp)

        peerConnection?.createAnswer(object : SimpleSdpObserver() {
            override fun onCreateSuccess(desc: SessionDescription?) {
                peerConnection?.setLocalDescription(SimpleSdpObserver(), desc)
                val payload = JSONObject()
                payload.put("type", "answer")
                payload.put("sdp", desc?.description)
                signalingClient.send(payload)
            }
        }, MediaConstraints())
    }

    private fun handleAnswer(json: JSONObject) {
        val sdp = SessionDescription(SessionDescription.Type.ANSWER, json.getString("sdp"))
        peerConnection?.setRemoteDescription(SimpleSdpObserver(), sdp)
    }

    private fun handleCandidate(json: JSONObject) {
        val candidate = IceCandidate(
            json.getString("sdpMid"),
            json.getInt("sdpMLineIndex"),
            json.getString("candidate")
        )
        peerConnection?.addIceCandidate(candidate)
    }

    fun sendData(data: ByteArray) {
        if (dataChannel?.state() == DataChannel.State.OPEN) {
            val buffer = DataChannel.Buffer(ByteBuffer.wrap(data), true)
            dataChannel?.send(buffer)
        }
    }
}

package com.ug.phyfi_router.protocol

/**
 * Utilitário estrito para serialização e deserialização do protocolo binário.
 * Estrutura: [4-bits Actor ID] + [4-bits Timestamp] + [8-bits Intent] = 16 bits.
 */
object Protocol16Bit {
    
    fun pack(actorId: Byte, timestamp: Byte, intent: Byte): Short {
        val a = (actorId.toInt() and 0x0F) shl 12
        val t = (timestamp.toInt() and 0x0F) shl 8
        val i = intent.toInt() and 0xFF
        return (a or t or i).toShort()
    }

    fun unpack(payload: Short): PhyFiPayload {
        val data = payload.toInt()
        val actorId = ((data shr 12) and 0x0F).toByte()
        val timestamp = ((data shr 8) and 0x0F).toByte()
        val intent = (data and 0xFF).toByte()
        
        return PhyFiPayload(actorId, timestamp, intent)
    }
}

class PhyFiService : Service() {

    private lateinit var wakeLockManager: WakeLockManager
    private lateinit var foregroundNotifier: ForegroundNotifier

    private var signalingClient: SignalingClient? = null
    private var webRtcManager: WebRTCManager? = null
    private var audioCaptureManager: AudioCaptureManager? = null
    private var audioDecoder: AudioDecoder? = null
    private var bluetoothManager: BluetoothHidManager? = null
    private var kineFiTranslator: KineFiTranslator? = null

    private val serviceScope = CoroutineScope(Dispatchers.IO)
    private var activePin: String? = null

    override fun onCreate() {
        super.onCreate()
        wakeLockManager = WakeLockManager(this).apply { acquire() }
        foregroundNotifier = ForegroundNotifier(this)

        try {
            val notification = foregroundNotifier.build(null)
            if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.Q) {
                var serviceType = ServiceInfo.FOREGROUND_SERVICE_TYPE_MEDIA_PLAYBACK
                if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.R) {
                    serviceType = serviceType or ServiceInfo.FOREGROUND_SERVICE_TYPE_MICROPHONE
                }
                startForeground(foregroundNotifier.notificationId, notification, serviceType)
            } else {
                startForeground(foregroundNotifier.notificationId, notification)
            }
        } catch (e: Exception) {
            logAsync("ERRO CRÍTICO no Foreground: ${e.message}")
            shutdownService()
        }
    }

    override fun onStartCommand(intent: Intent?, flags: Int, startId: Int): Int {
        when (intent?.action) {
            "STOP" -> {
                shutdownService()
                return START_NOT_STICKY
            }
            "INIT_HARDWARE" -> {
                if (bluetoothManager == null) initializeBluetoothStack()
            }
            "START_NETWORK" -> {
                val pin = intent.getStringExtra("ROOM_PIN")
                if (pin != null && activePin != pin) {
                    activePin = pin
                    initializeNetworkStack(pin)
                    initializeAudioStack()
                    foregroundNotifier.update(pin)
                }
            }
        }
        return START_STICKY
    }

    override fun onBind(intent: Intent?): IBinder? = null

    override fun onTaskRemoved(rootIntent: Intent?) {
        shutdownService()
        super.onTaskRemoved(rootIntent)
    }

    private fun initializeBluetoothStack() {
        bluetoothManager = BluetoothHidManager(applicationContext).apply { initialize() }
        kineFiTranslator = KineFiTranslator(bluetoothManager!!)
        logAsync("Kine-Fi ativado. Conecte o PC ao celular.")
    }

    private fun initializeNetworkStack(roomPin: String) {
        serviceScope.launch {
            try {
                // Alocação local garante imutabilidade para o compilador do Kotlin
                val currentSigClient =
                        SignalingClient(roomPin, { msg -> logAsync(msg) }) { jsonMessage ->
                            webRtcManager?.processSignalingMessage(jsonMessage)
                        }
                signalingClient = currentSigClient

                webRtcManager =
                        WebRTCManager(applicationContext, currentSigClient) { data ->
                            if (data.size >= 2) {
                                val high = data[0].toInt() and 0xFF
                                val low = data[1].toInt() and 0xFF
                                val payload = Protocol16Bit.unpack(((high shl 8) or low).toShort())
                                kineFiTranslator?.processIncomingPayload(payload)
                            }
                        }

                currentSigClient.connect()
                webRtcManager?.createLocalDataChannel()
            } catch (e: Exception) {
                logAsync("ERRO DE REDE: ${e.message}")
            }
        }
    }
    private fun initializeAudioStack() {
        audioDecoder =
                AudioDecoder(44100f) { payload ->
                    val networkPacket = ByteArray(2)
                    networkPacket[0] = (payload.toInt() shr 8).toByte()
                    networkPacket[1] = payload.toByte()
                    webRtcManager?.sendData(networkPacket)
                }

        audioCaptureManager = AudioCaptureManager { buffer -> audioDecoder?.processAudio(buffer) }
        audioCaptureManager?.startCapture()
        logAsync("So-Fi Acústico ativo. Aguardando Dicionário.")
    }

    private fun logAsync(msg: String) {
        serviceScope.launch { RouterState.log(msg) }
    }

    private fun shutdownService() {
        audioCaptureManager?.stopCapture()
        signalingClient?.disconnect()
        bluetoothManager?.shutdown()
        foregroundNotifier.release()
        wakeLockManager.release()

        sendBroadcast(Intent("com.ug.phyfi_router.SERVICE_STOPPED"))

        if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.N) {
            stopForeground(STOP_FOREGROUND_REMOVE)
        } else {
            @Suppress("DEPRECATION") stopForeground(true)
        }
        stopSelf()
    }
}

import android.content.Context

class WakeLockManager(context: Context) {
    private val powerManager = context.getSystemService(Context.POWER_SERVICE) as PowerManager
    private var wakeLock: PowerManager.WakeLock? = null

    fun acquire() {
        if (wakeLock == null) {
            wakeLock =
                    powerManager.newWakeLock(PowerManager.PARTIAL_WAKE_LOCK, "PhyFi::DSPWakeLock")
        }
        wakeLock?.acquire(10 * 60 * 1000L)
    }

    fun release() {
        if (wakeLock?.isHeld == true) {
            wakeLock?.release()
        }
    }
}

class MainActivity : AppCompatActivity() {

    private lateinit var routerUI: RouterUI
    private lateinit var permissionManager: PermissionManager
    private var hasHardwarePermissions = false

    private val enableBluetoothLauncher =
            registerForActivityResult(ActivityResultContracts.StartActivityForResult()) {
                checkBluetoothAndStartHardware()
            }

    private val serviceStoppedReceiver =
            object : BroadcastReceiver() {
                override fun onReceive(context: Context?, intent: Intent?) {
                    RouterState.clear()
                    routerUI.clearLogs()
                    routerUI.setReadyState(false)
                    routerUI.log("Serviço de roteamento finalizado.")
                }
            }

    override fun onCreate(savedInstanceState: Bundle?) {
        super.onCreate(savedInstanceState)

        routerUI = RouterUI(this)
        setContentView(routerUI.layout)
        permissionManager = PermissionManager(this)

        val filter = IntentFilter("com.ug.phyfi_router.SERVICE_STOPPED")
        if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.TIRAMISU) {
            registerReceiver(serviceStoppedReceiver, filter, Context.RECEIVER_NOT_EXPORTED)
        } else {
            @Suppress("UnspecifiedRegisterReceiverFlag")
            registerReceiver(serviceStoppedReceiver, filter)
        }

        bindUI()
        observeState()

        if (isServiceActivelyRunning()) {
            routerUI.log("Vinculado à sessão ativa no nível do SO.")
            hasHardwarePermissions = true
        } else {
            RouterState.clear()
            routerUI.clearLogs()
            requestHardwarePermissionsOnBoot()
        }
    }

    private fun requestHardwarePermissionsOnBoot() {
        permissionManager.checkAndRequest { allGranted ->
            hasHardwarePermissions = allGranted
            if (allGranted) {
                checkBluetoothAndStartHardware()
            } else {
                routerUI.log("AVISO: Permissões pendentes. O Roteador não funcionará.")
            }
        }
    }

    private fun checkBluetoothAndStartHardware() {
        val bluetoothAdapter = BluetoothAdapter.getDefaultAdapter()
        if (bluetoothAdapter?.isEnabled == false) {
            val enableBtIntent = Intent(BluetoothAdapter.ACTION_REQUEST_ENABLE)
            enableBluetoothLauncher.launch(enableBtIntent)
            return
        }
        startHardwareService()
    }

    private fun bindUI() {
        routerUI.connectBtn.setOnClickListener {
            if (!hasHardwarePermissions) {
                requestHardwarePermissionsOnBoot()
                return@setOnClickListener
            }

            val pin = routerUI.pinInput.text.toString().trim()
            if (pin.length < 4) {
                Toast.makeText(this, "Insira um PIN válido", Toast.LENGTH_SHORT).show()
                return@setOnClickListener
            }
            routerUI.setRoutingActiveState()
            startNetworkService(pin)
        }
    }

    private fun observeState() {
        lifecycleScope.launch { RouterState.logs.collect { msg -> routerUI.log(msg) } }
        lifecycleScope.launch {
            RouterState.isWebRtcConnected.collect { active -> routerUI.setWebRtcLed(active) }
        }
        lifecycleScope.launch {
            RouterState.isBluetoothConnected.collect { active ->
                routerUI.setBluetoothLed(active)
                routerUI.setReadyState(active)
            }
        }
    }

    @Suppress("DEPRECATION")
    private fun isServiceActivelyRunning(): Boolean {
        val manager = getSystemService(Context.ACTIVITY_SERVICE) as ActivityManager
        for (service in manager.getRunningServices(Int.MAX_VALUE)) {
            if (PhyFiService::class.java.name == service.service.className) return true
        }
        return false
    }

    private fun startHardwareService() {
        val serviceIntent =
                Intent(this, PhyFiService::class.java).apply { action = "INIT_HARDWARE" }
        startForegroundServiceCompat(serviceIntent)
    }

    private fun startNetworkService(pin: String) {
        val serviceIntent =
                Intent(this, PhyFiService::class.java).apply {
                    action = "START_NETWORK"
                    putExtra("ROOM_PIN", pin)
                }
        startForegroundServiceCompat(serviceIntent)
        lifecycleScope.launch { RouterState.log("Iniciando ponte de rede com PIN: $pin") }
    }

    private fun startForegroundServiceCompat(intent: Intent) {
        if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.O) {
            startForegroundService(intent)
        } else {
            startService(intent)
        }
    }

    override fun onDestroy() {
        super.onDestroy()
        unregisterReceiver(serviceStoppedReceiver)
    }

package ug.network.phyfi;

import ug.core.ClientEngine;
import ug.audio.AudioChannel;

class PhyFiAudioDispatcher {
	var engine:ClientEngine;
	var soundQueue:Array<String>;
	var queueIndex:Int;
	var timer:Float;
	var interval:Float;

	public function new(engine:ClientEngine) {
		this.engine = engine;
		this.soundQueue = new Array<String>();
		this.queueIndex = 0;
		this.timer = 0.0;
		this.interval = 0.06; // Cofre de 60ms estrito da FFT
	}

	public function enqueueSequence(sequence:Array<String>):Void {
		var len = sequence.length;
		for (i in 0...len) {
			this.soundQueue.push(sequence[i]);
		}
	}

	public function update(deltaTime:Float):Void {
		if (this.queueIndex >= this.soundQueue.length) {
			if (this.soundQueue.length > 0) {
				this.clear();
			}
			return;
		}

		this.timer -= deltaTime;
		if (this.timer <= 0.0) {
			this.timer = this.interval;
			var nextSound = this.soundQueue[this.queueIndex];
			this.queueIndex++;

			if (nextSound != null) {
				var channelInt:Int = cast AudioChannel.SFX;
				this.engine.audio.play2D(nextSound, cast channelInt, false, 1.0);
			}
		}
	}

	public function clear():Void {
		for (_while_ in 0...999999) {
			if (!(this.soundQueue.length > 0)) break;
			this.soundQueue.pop();
		}
		this.queueIndex = 0;
		this.timer = 0.0;
	}
}

package ug.network.phyfi;

class PhyFiAudioEncoder {
	public function new() {}

	public function encodePacket(actorId:Int, timestamp:Int, intent:Int):Array<String> {
		var a = (actorId & 0x0F) << 12;
		var t = (timestamp & 0x0F) << 8;
		var i = intent & 0xFF;
		var payload = a | t | i;

		return encodePayload(payload);
	}

	private function encodePayload(payload:Int):Array<String> {
		var sequence = new Array<String>();

		sequence.push(PhyFiSoundbank.HEADER);

		var n1 = (payload >> 12) & 0x0F;
		var n2 = (payload >> 8) & 0x0F;
		var n3 = (payload >> 4) & 0x0F;
		var n4 = payload & 0x0F;

		sequence.push(PhyFiSoundbank.getHex(n1));
		sequence.push(PhyFiSoundbank.getHex(n2));
		sequence.push(PhyFiSoundbank.getHex(n3));
		sequence.push(PhyFiSoundbank.getHex(n4));

		sequence.push(PhyFiSoundbank.TAIL);

		return sequence;
	}
}

package ug.network;

import ug.core.memory.MemoryPool;
import ug.network.NetworkManager;
import ug.network.OpCode;
import ug.network.Packet;
import ug.tools.Profiler;
import haxe.ds.Vector;

class ClientSimulation {
	var network:NetworkManager;
	var pool:MemoryPool;

	public var localPlayerId:Int;

	var currentTick:Int = 0;
	var lastReceivedServerTick:Int = -1;
	var activeFromServer:Vector<Bool>;

	public function new(net:NetworkManager, memoryPool:MemoryPool) {
		network = net;
		pool = memoryPool;
		localPlayerId = -1;
		activeFromServer = new Vector<Bool>(MemoryPool.MAX_ENTITIES);

		for (i in 0...MemoryPool.MAX_ENTITIES) {
			activeFromServer[i] = false;
		}

		network.router.subscribe(OpCode.SYNC_STATE, onSnapshotReceived);
		network.router.subscribe(OpCode.SYNC_FULL_STATE, onFullSnapshotReceived);
		network.router.subscribe(OpCode.NET_SPAWN, onNetSpawn);
		network.router.subscribe(OpCode.NET_DESPAWN, onNetDespawn);
		network.router.subscribe(OpCode.NET_PLAY_SFX, onNetPlaySfx);
		network.router.subscribe(OpCode.NET_SET_BGM, onNetSetBgm);
	}

	public function gatherAndSendInput(dirX:Float, dirZ:Float, yaw:Float, pitch:Float, buttons:Int, soundDb:Float = -99.0):Void {
		currentTick++;
		var payload = currentTick + ":" + dirX + ":" + dirZ + ":" + yaw + ":" + pitch + ":" + buttons + ":" + formatFloat(soundDb);
		network.sendToServer(new Packet(OpCode.INPUT_ACTION, payload));
	}

	inline function formatFloat(v:Float):Float {
		return Math.ffloor(v * 1000.0) / 1000.0;
	}

	function onFullSnapshotReceived(senderId:Int, packet:Packet):Void {
		Profiler.microCalls[Profiler.TRC_NET]++;

		var mainParts = packet.payload.split(";");
		if (mainParts.length < 2) {
			return;
		}

		var serverTick = Std.parseInt(mainParts[0]);
		lastReceivedServerTick = serverTick;

		for (i in 0...MemoryPool.MAX_ENTITIES) {
			activeFromServer[i] = false;
		}

		var entities = mainParts[1].split("|");
		for (entData in entities) {
			if (entData == "") {
				continue;
			}
			parseAndApplyEntity(entData, true);
		}

		for (i in 0...MemoryPool.MAX_ENTITIES) {
			if (activeFromServer[i]) {
				pool.entity_visible[i] = true;
			}
		}

		if (ug.core.ClientEngine.instance != null && ug.core.ClientEngine.instance.transformInterpolator != null) {
			ug.core.ClientEngine.instance.transformInterpolator.registerSnapshot();
		}
	}

	function onSnapshotReceived(senderId:Int, packet:Packet):Void {
		Profiler.microCalls[Profiler.TRC_NET]++;

		var mainParts = packet.payload.split(";");
		if (mainParts.length < 2) {
			return;
		}

		var serverTick = Std.parseInt(mainParts[0]);

		if (serverTick < lastReceivedServerTick) {
			return;
		}

		var isLargeGap = false;
		if (lastReceivedServerTick != -1) {
			var tickDiff = serverTick - lastReceivedServerTick;
			if (tickDiff > 10) {
				isLargeGap = true;
			}
		}

		lastReceivedServerTick = serverTick;

		var entities = mainParts[1].split("|");
		for (entData in entities) {
			if (entData == "") {
				continue;
			}
			parseAndApplyEntity(entData, isLargeGap);
		}

		for (i in 0...MemoryPool.MAX_ENTITIES) {
			if (pool.entity_active[i] && pool.is_networked[i]) {
				pool.entity_visible[i] = true;
			}
		}

		if (ug.core.ClientEngine.instance != null && ug.core.ClientEngine.instance.transformInterpolator != null) {
			ug.core.ClientEngine.instance.transformInterpolator.registerSnapshot();
		}
	}

	function parseAndApplyEntity(entData:String, isLargeGap:Bool):Void {
		var p = entData.split(",");
		if (p.length < 6) return;

		var id = Std.parseInt(p[0]);
		if (id < 0 || id >= MemoryPool.MAX_ENTITIES) {
			return;
		}

		activeFromServer[id] = true;
		pool.is_networked[id] = true;

		if (!pool.entity_active[id]) {
			pool.entity_active[id] = true;
			pool.entity_visible[id] = true;
			pool.is_static[id] = false;
			pool.transform.resetEntity(id);
			isLargeGap = true;
		}

		if (p.length >= 7 && p[p.length - 1] == "2D") {
			// 2D Payload: id, px, py, mask, isColliding, ai_state, "2D"
			var px = Std.parseFloat(p[1]);
			var py = Std.parseFloat(p[2]);
			
			pool.is_2d_entity[id] = true;
			
			var stateMask = Std.parseInt(p[3]);
			pool.entityState.stateMask[id] = stateMask;
			pool.isCollidingWall[id] = (p[4] == "1");
			pool.ai_state[id] = Std.parseInt(p[5]);

			pool.transform.localPosX[id] = px;
			pool.transform.localPosY[id] = py;
			
			// Emulação de isGrounded via flag bit 30
			pool.isGrounded[id] = ((stateMask & (1 << 30)) != 0);
		} else {
			// 3D Payload
			var px = Std.parseFloat(p[1]);
			var py = Std.parseFloat(p[2]);
			var pz = Std.parseFloat(p[3]);
			var yaw = Std.parseFloat(p[4]);
			var pitch = Std.parseFloat(p[5]);

			var stateMask = 0;
			if (p.length >= 7) {
				stateMask = Std.parseInt(p[6]);
			}

			pool.transform.pushHistoryState(id, px, py, pz, yaw, pitch);
			pool.transform.localPosX[id] = px;
			pool.transform.localPosY[id] = py;
			pool.transform.localPosZ[id] = pz;
			pool.transform.localRotYaw[id] = yaw;
			pool.transform.localRotPitch[id] = pitch;

			pool.entityState.stateMask[id] = stateMask;

			if (p.length >= 8) {
				pool.isCollidingWall[id] = (p[7] == "1");
			}
			if (p.length >= 9) {
				pool.ai_state[id] = Std.parseInt(p[8]);
			}

			if (isLargeGap) {
				pool.transform.localPosX[id] = px;
				pool.transform.localPosY[id] = py;
				pool.transform.localPosZ[id] = pz;
				pool.transform.localRotYaw[id] = yaw;
				pool.transform.localRotPitch[id] = pitch;
			}
		}
	}

	function onNetSpawn(senderId:Int, packet:Packet):Void {
		var parts = packet.payload.split(":");
		if (parts.length >= 5) {
			var id = Std.parseInt(parts[0]);
			var prefabId = parts[1];
			var px = Std.parseFloat(parts[2]);
			var py = Std.parseFloat(parts[3]);
			var pz = Std.parseFloat(parts[4]);

			var engine = ug.core.ClientEngine.instance;

			pool.entity_active[id] = true;

			ug.scene.PrefabLoader.instantiate(engine.core, engine.assets, engine.rasterizer, prefabId, px, py, pz, id, false);

			pool.entity_visible[id] = true;
			activeFromServer[id] = true;
			pool.is_networked[id] = true;
		}
	}

	function onNetDespawn(senderId:Int, packet:Packet):Void {
		var id = Std.parseInt(packet.payload);
		if (id != null) {
			pool.destroyEntity(id);
			activeFromServer[id] = false;
			pool.is_networked[id] = false;

			var engine = ug.core.ClientEngine.instance;
			if (engine != null && engine.rasterizer != null) {
				engine.rasterizer.entity_modelId[id] = -1;
				engine.rasterizer.entity_textureId[id] = -1;
			}
		}
	}

	function onNetPlaySfx(senderId:Int, packet:Packet):Void {
		var audioId = packet.payload;
		var engine = ug.core.ClientEngine.instance;
		engine.audio.play2D(audioId, ug.audio.AudioChannel.SFX, false, 1.0);
	}

	function onNetSetBgm(senderId:Int, packet:Packet):Void {
		var audioId = packet.payload;
		var engine = ug.core.ClientEngine.instance;
		engine.audio.setBGM(audioId, 1.0);
		if (engine.core != null && engine.core.memoryPool != null) {
			engine.core.memoryPool.globalBGM = audioId;
		}
	}
}

package ug.network;

/**
 * Atua como o "Barramento de Eventos" (Event Bus) da arquitetura.
 * Usa um Array pré-alocado no lugar de um Map para evitar quebras de Metatable no Luau,
 * garantindo performance O(1) absoluta (Data-Oriented Design).
 */
class NetworkRouter {
	/**
	 * O indice externo do Array representa o valor (Int) do OpCode.
	 * O array interno contem as funcoes que estao escutando aquele OpCode.
	 */
	var listeners:Array<Array<(Int, Packet) -> Void>>;

	public function new() {
		listeners = new Array<Array<(Int, Packet) -> Void>>();

		// Pre-aloca espaco para os OpCodes para previnir GC Spikes
		for (i in 0...20) {
			listeners.push(new Array<(Int, Packet) -> Void>());
		}
	}

	public function subscribe(opCode:OpCode, callback:(Int, Packet) -> Void):Void {
		var index:Int = cast opCode;

		// Seguranca: Expande o array dinamicamente se criarmos mais de 20 OpCodes no futuro
		for (_while_ in 0...999999) {
			if (!(listeners.length <= index)) break;
			listeners.push(new Array<(Int, Packet) -> Void>());
		}

		listeners[index].push(callback);
	}

	public function unsubscribe(opCode:OpCode, callback:(Int, Packet) -> Void):Void {
        var index:Int = cast opCode;

        // Verifica se o índice existe dentro dos limites do Array para evitar crashes
        if (index >= 0 && index < listeners.length) {
            listeners[index].remove(callback);
        }
    }

	public function handleServerPacket(senderId:Int, packet:Packet):Void {
		dispatch(senderId, packet);
	}

	public function handleClientPacket(packet:Packet):Void {
		dispatch(0, packet);
	}

	private function dispatch(senderId:Int, packet:Packet):Void {
		var index:Int = cast packet.opCode;

		// Verificacao de limites para garantir que pacotes hackers nao crashem o servidor
		if (index >= 0 && index < listeners.length) {
			var callbacks = listeners[index];
			// Roda todas as funcoes dos sistemas que pediram para ouvir essa mensagem
			for (callback in callbacks) {
				callback(senderId, packet);
			}
		}
	}
}

class ServerSimulation {
	var network:NetworkManager;
	var core:ug.core.Core;
	var pool:MemoryPool;

	var playerToEntity:IntMap<Int>;
	var currentTick:Int;
	var maxEntities:Int;

	var timeoutLimit:Int = cast(1.0 / RenderConfig.RENDER_CAP_MS);

	var entityLastInputTick:haxe.ds.Vector<Int>;

	public function new(net:NetworkManager, core:ug.core.Core, maxEnt:Int) {
		this.network = net;
		this.core = core;
		this.pool = core.memoryPool;
		maxEntities = maxEnt;
		playerToEntity = new IntMap<Int>();
		currentTick = 0;

		entityLastInputTick = new haxe.ds.Vector<Int>(maxEntities);
		for (i in 0...maxEntities) {
			entityLastInputTick[i] = 0;
		}

		network.router.subscribe(OpCode.INPUT_ACTION, onInputReceived);
		network.router.subscribe(OpCode.TELEPORT_INTENT, onTeleportReceived);

	public function assignPlayerEntity(playerId:Int, entityId:Int):Void {
		playerToEntity.set(playerId, entityId);
		entityLastInputTick[entityId] = currentTick;
	}

	function onTeleportReceived(senderId:Int, packet:Packet):Void {
		if (!playerToEntity.exists(senderId)) return;
		var entId = playerToEntity.get(senderId);
		if (!pool.entity_active[entId]) return;
		
		var parts = packet.payload.split(":");
		if (parts.length >= 5) {
			var px = Std.parseFloat(parts[0]);
			var py = Std.parseFloat(parts[1]);
			var pz = Std.parseFloat(parts[2]);
			var yaw = Std.parseFloat(parts[3]);
			var pitch = Std.parseFloat(parts[4]);
			
			pool.transform.localPosX[entId] = px;
			pool.transform.localPosY[entId] = py;
			pool.transform.localPosZ[entId] = pz;
			
			pool.transform.localRotYaw[entId] = yaw;
			pool.transform.localRotPitch[entId] = pitch;
			
			pool.velX[entId] = 0.0;
			pool.velY[entId] = 0.0;
			pool.velZ[entId] = 0.0;

			pool.is_dirty[entId] = true;
		}
	}

	function onInputReceived(senderId:Int, packet:Packet):Void {
		if (pool.velY[0] == 99999.0)
			return;

		if (!playerToEntity.exists(senderId))
			return;
		var entId = playerToEntity.get(senderId);
		if (!pool.entity_active[entId])
			return;

		entityLastInputTick[entId] = currentTick;

		var parts = packet.payload.split(":");
		if (parts.length >= 6) {
			var dirX = Std.parseFloat(parts[1]);
			var dirZ = Std.parseFloat(parts[2]);
			var yaw = Std.parseFloat(parts[3]);
			var pitch = Std.parseFloat(parts[4]);
			var buttons = Std.parseInt(parts[5]);

			if (pool.is_2d_entity[entId]) {
				var walkSpeed = 200.0;
				pool.vel2d_x[entId] = dirX * walkSpeed;
				
				if (buttons != null) {
					pool.entityState.stateMask[entId] = buttons;
					if ((buttons & 1) != 0) {
						if (pool.isGrounded[entId]) {
							pool.vel2d_y[entId] = -400.0;
							pool.isGrounded[entId] = false;
						}
					}
				}
				pool.is_dirty[entId] = true;
			} else {
				var walkSpeed = 12.0;
				pool.velX[entId] = dirX * walkSpeed;
				pool.velZ[entId] = dirZ * walkSpeed;
				pool.transform.localRotYaw[entId] = yaw;
				pool.transform.localRotPitch[entId] = pitch;

				pool.is_dirty[entId] = true;

				if (buttons != null) {
					pool.entityState.stateMask[entId] = buttons;

					if ((buttons & 1) != 0) {
						if (pool.isGrounded[entId]) {
							pool.velY[entId] = 12.0;
							pool.isGrounded[entId] = false;
						}
					}
				}
			}
			
			if (parts.length >= 7) {
				var soundDb = Std.parseFloat(parts[6]);
				if (soundDb > -90.0) {
					this.core.eventBus.pushEvent(ug.core.EventBus.EVT_SOUND_EMITTED, entId, 0, 0.0, soundDb);
				}
			}
		}
	}

	public function processTick():Void {
		currentTick++;

		for (i in 0...maxEntities) {
			if (pool.entity_active[i] && !pool.is_static[i]) {
				var isPlayer = false;
				for (key in playerToEntity.keys()) {
					if (playerToEntity.get(key) == i) {
						isPlayer = true;
						break;
					}
				}

				if (isPlayer && entityLastInputTick[i] != 0 && (currentTick - entityLastInputTick[i]) > timeoutLimit) {
					if (pool.is_2d_entity[i]) {
						pool.vel2d_x[i] = 0.0;
					} else {
						pool.velX[i] = 0.0;
						pool.velZ[i] = 0.0;
					}
					pool.entityState.stateMask[i] = 0;
					pool.is_dirty[i] = true;
				}
			}
		}
	}

package ug.network;

import ug.tools.Logger;

class SessionManager {
	public static inline var MAX_PLAYERS:Int = 4;

	public static var isMatchInProgress:Bool = false;
	public static var isHostLocal:Bool = false;

	public static var localPlayerId:Int = -1;
	public static var roomPin:String = "";

	public static var activePlayers:Array<Int> = new Array<Int>();

	public static var isPlayerReady:Array<Bool> = new Array<Bool>();
	public static var playerToNetworkId:Array<Int> = new Array<Int>();

	public static function init():Void {
		if (isPlayerReady.length == 0) {
			for (i in 0...500) {
				isPlayerReady.push(false);
				playerToNetworkId.push(-1);
			}
		}
	}

	public static function tryJoinParty(playerId:Int, networkId:Int):Bool {
		if (activePlayers.indexOf(playerId) != -1) {
			return true;
		}

		if (activePlayers.length >= MAX_PLAYERS) {
			Logger.warn("Sala cheia! Jogador " + playerId + " foi rejeitado.");
			return false;
		}

		activePlayers.push(playerId);
		isPlayerReady[playerId] = false;
		playerToNetworkId[playerId] = networkId;

		Logger.info("Jogador " + playerId + " entrou na Party! (" + activePlayers.length + "/" + MAX_PLAYERS + ")");
		return true;
	}

	public static function isPlayerInParty(playerId:Int):Bool {
		return activePlayers.indexOf(playerId) != -1;
	}

	public static function disbandParty():Void {
		for (_while_ in 0...999999) {
			if (!(activePlayers.length > 0)) break;
			activePlayers.pop();
		}
		isMatchInProgress = false;
		isHostLocal = false;
		localPlayerId = -1;
		roomPin = "";

		for (i in 0...500) {
			if (i < isPlayerReady.length) {
				isPlayerReady[i] = false;
				playerToNetworkId[i] = -1;
			}
		}
	}

	public static function hardReset():Void {
		disbandParty();
	}
}

			var data:Dynamic = Json.parse(e.data);
			if (data.room != this.room) {
				return;
			}

			if (this.isHost) {
				if (data.type == "join" && data.isHost == false) {
					this.listener.onSignalerPeerJoin();
				} else if (data.type == "answer") {
					this.listener.onSignalerAnswer(data.clientId, data.sdp);
				} else if (data.type == "ice" && data.isHost == false) {
					this.listener.onSignalerIceCandidate(data.clientId, false, data.candidate, data.sdpMid, data.sdpMLineIndex);
				}
			} else {
				if (data.type == "offer") {
					if (this.assignedClientId != -1) {
						return; // Correção: Bloqueia ofertas de WebRTC subsequentes destinadas a terceiros
					}
					this.assignedClientId = data.clientId;
					this.listener.onSignalerOffer(data.clientId, data.sdp);
				} else if (data.type == "ice" && data.isHost == true) {
					if (data.clientId != this.assignedClientId) {
						return; // Correção: Ignora candidatos ICE não pertencentes a esta sessão
					}
					this.listener.onSignalerIceCandidate(0, true, data.candidate, data.sdpMid, data.sdpMLineIndex);
				}
			}
		};

		this.ws.onerror = function(e:Dynamic) {
			this.listener.onSignalerError();
		};

		this.ws.onclose = function(e:Dynamic) {
			this.listener.onSignalerClose();
		};
	}

	public function sendOffer(clientId:Int, sdp:String):Void {
		var state:Int = cast this.ws.readyState;
		if (state == WebSocket.OPEN) {
			var payload = {
				type: "offer",
				room: this.room,
				clientId: clientId,
				sdp: sdp
			};
			this.ws.send(Json.stringify(payload));
		}
	}

	public function sendAnswer(sdp:String):Void {
		var state:Int = cast this.ws.readyState;
		if (state == WebSocket.OPEN) {
			var payload = {
				type: "answer",
				room: this.room,
				clientId: this.assignedClientId,
				sdp: sdp
			};
			this.ws.send(Json.stringify(payload));
		}
	}

	public function sendIce(isHost:Bool, candidate:Dynamic, sdpMid:String, sdpMLineIndex:Int, targetClientId:Int = -1):Void {
		var state:Int = cast this.ws.readyState;
		if (state == WebSocket.OPEN) {
			var cId = this.assignedClientId;
			if (isHost) {
				cId = targetClientId;
			}
			var payload = {
				type: "ice",
				room: this.room,
				clientId: cId,
				isHost: isHost,
				candidate: candidate,
				sdpMid: sdpMid,
				sdpMLineIndex: sdpMLineIndex
			};
			this.ws.send(Json.stringify(payload));
		}
	}
}

import ug.network.INetworkProvider;
import ug.network.Packet;
import ug.tools.Logger;
import js.html.rtc.PeerConnection;
import js.html.rtc.DataChannel;
import js.html.rtc.SessionDescription;
import js.html.rtc.IceCandidate;

class WebRTCProvider implements INetworkProvider implements ISignalerListener {
	public static var localHostProvider:WebRTCProvider = null;
	public static var localClientReceiver:(Packet) -> Void = null;

	public var disconnectCallbackClient:Void->Void;
	public var connectCallbackClient:Void->Void;
	public var disconnectCallbackServer:Int->Void;

	public var isHost:Bool = false;
	public var room:String = "";
	public var signaler:WebRTCSignaler;

	public var receiveCallbackHost:(Int, Packet) -> Void;
	public var receiveCallbackClient:(Packet) -> Void;

	public var hostChannels:Array<DataChannel>;
	public var hostPeers:Array<PeerConnection>;
	public var clientChannel:DataChannel;
	public var clientPeer:PeerConnection;

	var isLocalClient:Bool = false;
	var localClientId:Int = -1;

	public function new() {
		this.hostChannels = new Array<DataChannel>();
		this.hostPeers = new Array<PeerConnection>();
	}

	public function startServer(roomHash:String, onReceiveCallback:(Int, Packet) -> Void):Void {
		this.isHost = true;
		this.room = roomHash;
		this.receiveCallbackHost = onReceiveCallback;

		WebRTCProvider.localHostProvider = this;
		WebRTCProvider.localClientReceiver = null;

		Logger.info("[WebRTC] Hosteando sala: " + roomHash);

		this.signaler = new WebRTCSignaler(this.room, true, this);
		this.signaler.connect();
	}

	public function connectClient(roomHash:String, onReceiveCallback:(Packet) -> Void):Void {
		this.isHost = false;
		this.room = roomHash;
		this.receiveCallbackClient = onReceiveCallback;

		if (WebRTCProvider.localHostProvider != null && WebRTCProvider.localHostProvider.room == roomHash) {
			Logger.info("[WebRTC] Loopback Local ativado. Host em 0ms.");
			this.isLocalClient = true;
			WebRTCProvider.localClientReceiver = this.receiveCallbackClient;

			var host = WebRTCProvider.localHostProvider;
			this.localClientId = host.hostChannels.length;
			host.hostChannels.push(null);
			host.hostPeers.push(null); // Correção: Mantém o array paralelo em sincronia com os canais

			if (this.connectCallbackClient != null) {
				this.connectCallbackClient();
			}
			return;
		}

		Logger.info("[WebRTC] Conectando na sala Remota: " + roomHash);

		this.signaler = new WebRTCSignaler(this.room, false, this);
		this.signaler.connect();
	}

	public function onSignalerPeerJoin():Void {
		var clientId = this.hostChannels.length; // Correção: Fonte de verdade absoluta para IDs

		this.hostPeers.push(null);
		this.hostChannels.push(null);

		var handler = new HostPeerHandler(this, clientId);
		handler.setupHostPeer();
	}

	public function onSignalerOffer(clientId:Int, sdp:String):Void {
		var handler = new ClientPeerHandler(this);
		handler.setupClientPeer(sdp);
	}

	public function onSignalerAnswer(clientId:Int, sdp:String):Void {
		var p = this.hostPeers[clientId];
		if (p != null) {
			var jsSession = new SessionDescription({type: js.html.rtc.SdpType.ANSWER, sdp: sdp});
			p.setRemoteDescription(jsSession);
		}
	}

	public function onSignalerIceCandidate(clientId:Int, isHost:Bool, candidate:Dynamic, sdpMid:String, sdpMLineIndex:Int):Void {
		var ice = new IceCandidate({candidate: candidate, sdpMid: sdpMid, sdpMLineIndex: sdpMLineIndex});

		if (this.isHost) {
			var p = this.hostPeers[clientId];
			if (p != null)
				p.addIceCandidate(ice);
		} else {
			if (this.clientPeer != null)
				this.clientPeer.addIceCandidate(ice);
		}
	}

	public function onSignalerError():Void {
		Logger.error("[WebRTC] Erro no servidor de sinalizacao.");
		if (this.disconnectCallbackClient != null)
			this.disconnectCallbackClient();
	}

	public function onSignalerClose():Void {
		Logger.warn("[WebRTC] Conexao sinalizadora encerrada.");
		if (this.disconnectCallbackClient != null)
			this.disconnectCallbackClient();
	}

	public function sendToServer(packet:Packet):Void {
		if (this.isLocalClient && WebRTCProvider.localHostProvider != null) {
			if (WebRTCProvider.localHostProvider.receiveCallbackHost != null) {
				WebRTCProvider.localHostProvider.receiveCallbackHost(this.localClientId, packet);
			}
			return;
		}
		if (this.clientChannel != null) {
			var state:String = cast this.clientChannel.readyState;
			if (state == "open") {
				this.clientChannel.send(packet.opCode + "|" + packet.payload);
			}
		}
	}

	public function broadcast(packet:Packet):Void {
		var dataStr = packet.opCode + "|" + packet.payload;
		for (i in 0...this.hostChannels.length) {
			var dc = this.hostChannels[i];
			if (dc == null) {
				if (WebRTCProvider.localClientReceiver != null) {
					WebRTCProvider.localClientReceiver(packet);
				}
			} else {
				var state:String = cast dc.readyState;
				if (state == "open")
					dc.send(dataStr);
			}
		}
	}

	public function sendToClient(clientId:Int, packet:Packet):Void {
		if (clientId >= 0 && clientId < this.hostChannels.length) {
			var dc = this.hostChannels[clientId];
			if (dc == null) {
				if (WebRTCProvider.localClientReceiver != null) {
					WebRTCProvider.localClientReceiver(packet);
				}
			} else {
				var state:String = cast dc.readyState;
				if (state == "open") {
					dc.send(packet.opCode + "|" + packet.payload);
				} else {
					Logger.warn("[WebRTC] Ignorado envio para Client " + clientId + " - State: " + state);
				}
			}
		}
	}

	public function parseData(rawData:String):Packet {
		var separatorIndex = rawData.indexOf("|");

		if (separatorIndex != -1) {
			var opCodeStr = rawData.substring(0, separatorIndex);
			var payloadStr = rawData.substring(separatorIndex + 1);

			var opCodeInt = Std.parseInt(opCodeStr);
			if (opCodeInt != null) {
				return new Packet(cast opCodeInt, payloadStr);
			}
		}

		return null;
	}

	public function setOnClientDisconnected(callback:Void->Void):Void {
		this.disconnectCallbackClient = callback;
	}

	public function setOnClientConnected(callback:Void->Void):Void {
		this.connectCallbackClient = callback;
	}

	public function setOnPeerDisconnected(callback:Int->Void):Void {
		this.disconnectCallbackServer = callback;
	}
}

import asyncio
import json
import os
import websockets

rooms = {}

async def handler(websocket):
    room_id = None
    try:
        async for message in websocket:
            data = json.loads(message)
            room_id = data.get("room")
            if not room_id:
                continue

            if room_id not in rooms:
                rooms[room_id] = set()
            rooms[room_id].add(websocket)

            for client in rooms[room_id]:
                if client != websocket:
                    await client.send(message)

    except websockets.exceptions.ConnectionClosed:
        pass
    finally:
        if room_id and room_id in rooms and websocket in rooms[room_id]:
            rooms[room_id].remove(websocket)
            if len(rooms[room_id]) == 0:
                del rooms[room_id]

async def main():
    # O provedor injetará a porta correta aqui. Caso falhe, usa 10000.
    port = int(os.environ.get("PORT", 10000))
    print(f"[UG Signaler] Servidor WebRTC ativo na porta {port}")
    
    # 0.0.0.0 permite acesso externo pelo balanceador de carga do Render
    async with websockets.serve(handler, "0.0.0.0", port):
        await asyncio.Future()

if __name__ == "__main__":
    asyncio.run(main())
